El archivo original no se modifica. */ -}} {{- $raw := .RawContent -}} {{- $lineas := split $raw "\n" -}} {{- if and .File (gt (len $lineas) 0) (strings.Contains (index $lineas 0) .File.ContentBaseName) -}} {{- $raw = delimit (after 1 $lineas) "\n" -}} {{- end -}} {{- return (.RenderString (dict "display" "block") $raw) -}}
